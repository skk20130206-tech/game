<style>
#orbit-share-button{display:inline-flex;align-items:center;justify-content:center;gap:7px}.orbit-share-toast{position:fixed;left:50%;bottom:26px;z-index:100005;transform:translate(-50%,18px);padding:11px 16px;border:1px solid #c6ead83b;border-radius:999px;background:#0c1c22ee;color:#eaf7ed;font:700 12px/1.2 Inter,Pretendard,system-ui,-apple-system,BlinkMacSystemFont,"Segoe UI",sans-serif;box-shadow:0 16px 50px #0009;opacity:0;pointer-events:none;transition:opacity .22s ease,transform .22s ease}.orbit-share-toast.show{opacity:1;transform:translate(-50%,0)}
</style>
<div id="orbit-share-toast" class="orbit-share-toast" role="status" aria-live="polite"></div>
<script>
(()=>{
  const guide=document.getElementById('orbit-start-guide');
  const actions=document.querySelector('.orbit-start-actions');
  const toast=document.getElementById('orbit-share-toast');
  if(!actions||document.getElementById('orbit-share-button'))return;

  const button=document.createElement('button');
  button.id='orbit-share-button';
  button.className='orbit-start-secondary';
  button.type='button';
  button.innerHTML='<span aria-hidden="true">↗</span> 친구에게 공유';
  if(guide&&guide.parentNode===actions)guide.insertAdjacentElement('afterend',button);else actions.appendChild(button);

  let toastTimer;
  const showToast=message=>{
    if(!toast)return;
    toast.textContent=message;
    toast.classList.add('show');
    clearTimeout(toastTimer);
    toastTimer=setTimeout(()=>toast.classList.remove('show'),2200);
  };

  const copyLink=async url=>{
    try{
      await navigator.clipboard.writeText(url);
      showToast('게임 링크를 복사했어요 ✓');
      return true;
    }catch(e){
      try{
        const input=document.createElement('textarea');
        input.value=url;
        input.setAttribute('readonly','');
        input.style.position='fixed';
        input.style.opacity='0';
        document.body.appendChild(input);
        input.select();
        const ok=document.execCommand('copy');
        input.remove();
        if(ok){showToast('게임 링크를 복사했어요 ✓');return true;}
      }catch(_){}
      showToast('주소창의 링크를 복사해 친구에게 보내주세요');
      return false;
    }
  };

  button.addEventListener('click',async()=>{
    const url=window.location.href.split('#')[0];
    const data={title:'ORBIT',text:'신비로운 행성을 탐험하는 ORBIT을 같이 해봐!',url};
    try{
      if(navigator.share){
        await navigator.share(data);
        showToast('공유했어요 ✓');
      }else{
        await copyLink(url);
      }
    }catch(error){
      if(error&&error.name==='AbortError')return;
      await copyLink(url);
    }
  });
})();
</script>
