  function creature(c,id,t=0,scale=1,met=false){
    const sp=SPECIES[id];if(!sp)return;c.save();c.scale(scale,scale);
    const bob=Math.sin(t*2.6)*2;
    ellipse(c,2,4,20,8,'#15252530');c.translate(0,-Math.abs(bob));
    if(sp.shape==='crab'){
      for(const side of[-1,1])for(let j=0;j<3;j++)line(c,[[side*12,-4+j*3],[side*(24+j*2),-5+j*6],[side*(29+j*2),2+j*4+Math.sin(t*3+j)]],sp.dark,4);
      ellipse(c,0,-13,23,16,sp.dark);ellipse(c,-1,-18,21,13,sp.color);ellipse(c,-8,-29,4,6,sp.color);ellipse(c,9,-29,4,6,sp.color);
      ellipse(c,-8,-30,2,3,'#243a3e');ellipse(c,9,-30,2,3,'#243a3e');
      for(const side of[-1,1]){line(c,[[side*16,-17],[side*27,-26]],sp.dark,4);ellipse(c,side*29,-30,8,6,sp.color);}
    }else{
      const floating=sp.shape==='float';if(floating)c.translate(0,-13-bob*2);
      if(sp.shape==='fluff'){ellipse(c,-14,-31,7,20,sp.dark);ellipse(c,13,-32,7,20,sp.color);}
      if(sp.shape==='flame'){
        polygon(c,[[-11,-26],[-9,-47],[0,-38],[7,-57],[15,-37],[13,-20]],'#ffa364');polygon(c,[[-4,-27],[3,-44],[10,-24]],'#ffe5a8');
      }
      if(sp.shape==='prism'){
        polygon(c,[[-23,-15],[-13,-35],[0,-43],[20,-27],[23,-5],[8,6],[-17,3]],sp.color);polygon(c,[[0,-43],[20,-27],[23,-5],[7,-12]],sp.dark);
      }else{ellipse(c,0,-13,23,25,sp.dark);ellipse(c,-2,-17,22,23,sp.color);ellipse(c,-10,-23,9,12,'#ffffff14');}
      if(sp.shape==='sprout'){
        line(c,[[0,-34],[0,-51]],'#699b73',3);ellipse(c,-7,-48,9,5,'#89c57b');ellipse(c,8,-54,11,6,'#c3ec83');
      }
      if(!floating){ellipse(c,-13,5,7,4,sp.dark);ellipse(c,12,5,7,4,sp.dark);}
      else{for(let i=0;i<3;i++)line(c,[[-10+i*10,4],[-11+i*10,15+Math.sin(t*3+i)*4]],'#b9f2e899',3);glow(c,0,-15,47,'#b1f7fd17');}
      ellipse(c,-9,-19,2.5,4,'#1d3b3e');ellipse(c,8,-19,2.5,4,'#1d3b3e');ellipse(c,-10,-20,1,1.3,'#ffffed');ellipse(c,7,-20,1,1.3,'#ffffed');
      line(c,[[-3,-9],[0,-7],[3,-9]],'#415c58',1.5);ellipse(c,-15,-11,4,2,'#efac8c55');ellipse(c,13,-11,4,2,'#efac8c55');
    }
    if(met){c.fillStyle='#d7f2ac';c.font='11px sans-serif';c.textAlign='center';c.fillText('♥',0,-65-bob);}
    c.restore();
  }
